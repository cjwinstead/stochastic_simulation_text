% preamble.tex: LaTeX additions to pandoc's default template for the PDF.
% Loaded from header-includes in book_info.yaml. Page layout, language,
% contents and lists are set in book_info.yaml.

% ---------------------------------------------------------------------------
% Fonts: DejaVu Serif for text; Everson Mono, else JuliaMono, for code.
%
% Each font is tried first by family name, which finds fonts installed in the
% system or user font folders (e.g. ~/.local/share/fonts, as fc-list shows).
% Otherwise it is loaded by file name, which finds the copies that ship with
% TeX Live and MiKTeX (packages "dejavu" and "juliamono") even when they are
% not registered with the system.
% ---------------------------------------------------------------------------
\IfFontExistsTF{DejaVu Serif}
  {\setmainfont{DejaVu Serif}}
  {\setmainfont{DejaVuSerif}[
     Extension      = .ttf,
     BoldFont       = DejaVuSerif-Bold,
     ItalicFont     = DejaVuSerif-Italic,
     BoldItalicFont = DejaVuSerif-BoldItalic]}
\IfFontExistsTF{Everson Mono}
  {\setmonofont{Everson Mono}}
  {\IfFontExistsTF{JuliaMono}
     {\setmonofont{JuliaMono}}
     {\setmonofont{JuliaMono}[
        Extension      = .ttf,
        UprightFont    = *-Regular,
        BoldFont       = *-Bold,
        ItalicFont     = *-RegularItalic,
        BoldItalicFont = *-BoldItalic]}}

% ---------------------------------------------------------------------------
% Draft text awaiting the author's review (see textbook.lua)
% ---------------------------------------------------------------------------
\definecolor{draftcolor}{RGB}{164,0,0}

% ---------------------------------------------------------------------------
% Code blocks. pandoc sets code in its Highlighting environment (fancyvrb).
% fvextra adds line breaking: code is set one size smaller, and a line that
% is still wider than the space available, such as inside an example box,
% wraps with a continuation mark instead of running past the margin.
% ---------------------------------------------------------------------------
\usepackage{fvextra}
\fvset{fontsize=\small}
\RecustomVerbatimEnvironment{Highlighting}{Verbatim}%
  {commandchars=\\\{\},breaklines,breaknonspaceingroup,breakanywhere}

% ---------------------------------------------------------------------------
% Mathematics and figures
% ---------------------------------------------------------------------------
\usepackage{mathtools}
% Under XeLaTeX, pandoc's template loads unicode-math, which does not support
% \boldsymbol. The text uses it only for Greek letters (bold random variables
% such as \boldsymbol{\xi}), so map it to unicode-math's bold italic letters.
% The Markdown keeps \boldsymbol, which MathJax needs for the HTML.
\AtBeginDocument{\renewcommand{\boldsymbol}[1]{\symbfit{#1}}}
\usepackage{graphicx}
\usepackage{tikz}
\usepackage{pgfplots}
\pgfplotsset{compat=1.18}   % current behaviour, not backwards-compatibility mode
\usetikzlibrary{arrows,calc,shapes,backgrounds,intersections,patterns,positioning}
\usepackage[american]{circuitikz}
\tikzset{plotfigstyle/.style={}}
\pgfplotsset{plotaxisstyle/.style={width=0.9\linewidth,height=0.6\linewidth},legend pos=south east}

% ---------------------------------------------------------------------------
% Cross-references. cleveref must load after hyperref, which pandoc's template
% loads after this preamble, so both it and the names that use it are deferred
% to the end of the preamble. pandoc-crossref defers its own setup the same
% way and loads cleveref only if it is not already loaded; this hook is
% registered first, so these options take effect.
% ---------------------------------------------------------------------------
\usepackage{etoolbox}
\AtEndPreamble{%
  \usepackage[capitalize,nameinlink,noabbrev]{cleveref}%
  \crefname{example}{example}{examples}%
  \Crefname{example}{Example}{Examples}%
  \crefname{source}{julia example}{julia examples}%
  \Crefname{source}{Julia Example}{Julia Examples}%
}

% ---------------------------------------------------------------------------
% Example boxes, produced by textbook.lua from ::: {.example} divs
% ---------------------------------------------------------------------------
\usepackage{tcolorbox}
\tcbuselibrary{theorems,breakable,skins}
% breakable: a box longer than the space left on a page continues on the next
% page instead of moving whole and leaving a gap. enhanced jigsaw draws the
% split edges cleanly.
\tcbset{exampleboxstyle/.style={breakable,enhanced jigsaw,parbox=false,
  colback=green!5,colframe=green!35!black,fonttitle=\bfseries}}
\newtcbtheorem[number within=chapter]{example}{Example}{exampleboxstyle}{th}
\newtcbtheorem[number within=chapter]{source}{Julia Example}{exampleboxstyle}{th}
\newcommand{\exampleautorefname}{Example}
\newcommand{\sourceautorefname}{Julia Example}

% ---------------------------------------------------------------------------
% Title page: large bold title, author in small capitals
% ---------------------------------------------------------------------------
\usepackage{titling}
\pretitle{\begin{center}\Huge\bfseries}
\posttitle{\par\end{center}\vskip 1em}
\preauthor{\begin{center}\large\scshape}
\postauthor{\par\end{center}}

% ---------------------------------------------------------------------------
% Chapter headings show the number without the word "Chapter"
% ---------------------------------------------------------------------------
\makeatletter
\renewcommand{\@chapapp}{}
\makeatother

% ---------------------------------------------------------------------------
% dedication environment (used in dedication.tex)
% Source: http://www.tug.org/pipermail/texhax/2010-June/015184.html
% ---------------------------------------------------------------------------
\newenvironment{dedication}
  {\cleardoublepage
   \thispagestyle{empty}
   \vspace*{\stretch{1}}
   \hfill\begin{minipage}[t]{0.66\textwidth}
   \raggedright}
  {\end{minipage}
   \vspace*{\stretch{3}}
   \clearpage}

% ---------------------------------------------------------------------------
% chapquote environment: a quotation at the start of a chapter
% Source: http://tex.stackexchange.com/a/53380
% ---------------------------------------------------------------------------
\makeatletter
\newenvironment{chapquote}[2][2em]
  {\setlength{\@tempdima}{#1}%
   \def\chapquote@author{#2}%
   \parshape 1 \@tempdima \dimexpr\textwidth-2\@tempdima\relax%
   \itshape}
  {\par\normalfont\hfill--\ \chapquote@author\hspace*{\@tempdima}\par\bigskip}
\makeatother
